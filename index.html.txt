// ---------- 충돌/이동 ----------
function collides(piece, offX, offY, cells) {
    const useCells = cells || piece.cells;
    for (const [cx, cy] of useCells) {
        const gx = piece.x + cx + offX;
        const gy = piece.y + cy + offY;
        if (gx < 0 || gx >= COLS || gy >= ROWS) return true;
        if (gy >= 0 && grid[gy][gx]) return true;
    }
    return false;
}

function moveLeft() { if (!running) return; if (!collides(current, -1, 0)) current.x--; draw(); }
function moveRight() { if (!running) return; if (!collides(current, 1, 0)) current.x++; draw(); }
function softDrop() { if (!running) return; if (!collides(current, 0, 1)) { current.y++; score += 1; updateHud(); } else lockPiece(); draw(); }
function hardDrop() {
    if (!running) return;
    let dist = 0;
    while (!collides(current, 0, dist+1)) dist++;
    current.y += dist;
    score += dist * 2;
    updateHud();
    lockPiece();
    draw();
}

function rotatePiece() {
    if (!running) return;
    if (current.key === 'O') return;
    const pivot = current.cells[ current.key === 'I' ? 1 : 2 ] || current.cells[1];
    const rotated = current.cells.map(([cx, cy]) => {
        const rx = pivot[0] - (cy - pivot[1]);
        const ry = pivot[1] + (cx - pivot[0]);
        return [rx, ry];
    });
    const kicks = [0, -1, 1, -2, 2];
    for (const k of kicks) {
        if (!collides(current, k, 0, rotated)) {
            current.cells = rotated;
            current.x += k;
            draw();
            return;
        }
    }
}

function doHold() {
    if (!running || !canHold) return;
    if (holdPiece === null) {
        holdPiece = current.key;
        spawnNext();
    } else {
        const tmp = holdPiece;
        holdPiece = current.key;
        current = createPiece(tmp);
    }
    canHold = false;
    drawHold();
    draw();
}

function spawnNext() {
    refillBagIfNeeded();
    current = createPiece(bag.shift());
    canHold = true;
    if (collides(current, 0, 0)) {
        triggerGameOver();
    }
    drawNext();
}

function lockPiece() {
    for (const [cx, cy] of current.cells) {
        const gx = current.x + cx, gy = current.y + cy;
        if (gy < 0) { triggerGameOver(); return; }
        grid[gy][gx] = current.color;
    }
    clearLines();
    spawnNext();
}

function clearLines() {
    let cleared = 0;
    for (let y = ROWS - 1; y >= 0; y--) {
        if (grid[y].every(c => c !== null)) {
            grid.splice(y, 1);
            grid.unshift(Array(COLS).fill(null));
            cleared++;
            y++;
        }
    }
    if (cleared > 0) {
        const points = [0, 100, 300, 500, 800][cleared] * level;
        score += points;
        lines += cleared;
        coins += cleared * COIN_PER_LINE;
        const newLevel = Math.floor(lines / 10) + 1;
        if (newLevel !== level) {
            level = newLevel;
            dropInterval = Math.max(120, 800 - (level - 1) * 65);
        }
        linesSinceQuiz += cleared;
        updateHud();
        if (linesSinceQuiz >= QUIZ_LINE_THRESHOLD) {
            triggerQuiz();
        }
    }
}

// ---------- 퀴즈 ----------
function triggerQuiz() {
    running = false;
    
    // 섞인 퀴즈 목록에서 순서대로 하나씩 출제
    if (currentQuizIndex >= shuffledQuizBank.length) {
        for (let i = shuffledQuizBank.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [shuffledQuizBank[i], shuffledQuizBank[j]] = [shuffledQuizBank[j], shuffledQuizBank[i]];
        }
        currentQuizIndex = 0;
    }
    currentQuiz = shuffledQuizBank[currentQuizIndex++];

    document.getElementById('quiz-question').innerText = currentQuiz.q;
    document.getElementById('quiz-feedback').innerText = '';
    document.getElementById('quiz-continue-btn').style.display = 'none';
    const optDiv = document.getElementById('quiz-options');
    optDiv.innerHTML = '';
    currentQuiz.opts.forEach((opt, i) => {
        const b = document.createElement('button');
        b.className = 'quiz-opt';
        b.innerText = opt;
        b.onclick = () => answerQuiz(i);
        optDiv.appendChild(b);
    });
    document.getElementById('quiz-overlay').classList.add('active');
}

function answerQuiz(i) {
    const correct = i === currentQuiz.answer;
    document.querySelectorAll('.quiz-opt').forEach((b, idx) => {
        b.disabled = true;
        if (idx === currentQuiz.answer) b.classList.add('correct');
        else if (idx === i && !correct) b.classList.add('wrong');
    });
    const fb = document.getElementById('quiz-feedback');
    if (correct) {
        coins += COIN_PER_QUIZ;
        fb.innerHTML = `<span style="color:#4dff88;">⭕ 정답! +${COIN_PER_QUIZ}코인 획득!</span>`;
    } else {
        addGarbageRow();
        fb.innerHTML = `<span style="color:#ff4d4d;">❌ 오답! 쓰레기 줄이 추가됩니다.</span>`;
    }
    updateHud();
    document.getElementById('quiz-continue-btn').style.display = 'inline-block';
}

function afterQuiz() {
    document.getElementById('quiz-overlay').classList.remove('active');
    linesSinceQuiz = 0;
    resetQuizTimer(); // 퀴즈 종료 시 랜덤 시간 타이머 리셋
    updateHud();
    if (!gameOver) {
        running = true;
        lastTime = performance.now();
        requestAnimationFrame(loop);
    }
    draw();
}

function removeBestRow() {
    let bestIdx = -1, bestCount = -1;
    for (let y = 0; y < ROWS; y++) {
        const count = grid[y].filter(c => c !== null).length;
        if (count > 0 && count < COLS && count > bestCount) { bestCount = count; bestIdx = y; }
    }
    if (bestIdx === -1) return;
    grid.splice(bestIdx, 1);
    grid.unshift(Array(COLS).fill(null));
}

function addGarbageRow() {
    grid.shift();
    const gapCol = Math.floor(Math.random() * COLS);
    const newRow = Array(COLS).fill('#555569');
    newRow[gapCol] = null;
    grid.push(newRow);
    if (collides(current, 0, 0)) triggerGameOver();
}

    document.getElementById('shop-overlay').classList.remove('active');

function renderShop() {
    document.getElementById('shop-coin-txt').innerText = coins;
    const list = document.getElementById('shop-list');
    list.innerHTML = '';
    SHOP_ITEMS.forEach(item => {
        const row = document.createElement('div');
        row.className = 'shop-row';
        row.innerHTML = `<div class="shop-row-info"><b>${item.name} (${item.cost}💎)</b><span>${item.desc}</span></div>`;
        const btn = document.createElement('button');
        btn.className = 'shop-buy-btn';
        btn.innerText = '구매';
        btn.disabled = coins < item.cost;
        btn.onclick = () => buyItem(item.id);
        row.appendChild(btn);
        list.appendChild(row);
    });
}

function buyItem(id) {
    const item = SHOP_ITEMS.find(i => i.id === id);
    if (!item || coins < item.cost) return;
    coins -= item.cost;
    inventory[id] = (inventory[id] || 0) + 1;
    updateHud();
    updateInventoryUI();
    renderShop();
}

function updateInventoryUI() {
    SHOP_ITEMS.forEach(item => {
        const count = inventory[item.id] || 0;
        document.getElementById('item-count-' + item.id).innerText = count;
        const box = document.getElementById('item-box-' + item.id);
        if (count > 0) box.classList.remove('empty'); else box.classList.add('empty');
    });
}

function useItem(id) {
    if (!running || gameOver) return;
    if (!inventory[id] || inventory[id] <= 0) return;
    inventory[id]--;
    if (id === 'clearLine') {
        removeBestRow();
    } else if (id === 'summonI') {
        bag.unshift('I');
        drawNext();
    } else if (id === 'slowDown') {
        slowEndMs = elapsedPlayMs + 15000;
    }
    updateInventoryUI();
    updateHud();
    draw();
}

// ---------- 리더보드 (로컬 스토리지) ----------
function updateLeaderboard(newScore, newLines) {
    let board = JSON.parse(localStorage.getItem('tetris_scores') || '[]');
    
    const currentRecord = { score: newScore, lines: newLines };
    board.push(currentRecord);
    
    board.sort((a, b) => b.score - a.score);
    board = board.slice(0, 5);
    
    localStorage.setItem('tetris_scores', JSON.stringify(board));
    
    let html = '<div class="leaderboard"><h3>🏆 TOP 5 스코어</h3><ol>';
    board.forEach((item) => {
        const isCurrent = (item === currentRecord);
        html += `<li class="${isCurrent ? 'new-record' : ''}">${item.score}점 (${item.lines}줄)</li>`;
    });
    html += '</ol></div>';
    
    return html;
}

function triggerGameOver() {
    gameOver = true; running = false;
    
    const leaderboardHtml = updateLeaderboard(score, lines);
    
    document.getElementById('overlay-title').innerText = '💥 게임 오버';
    document.getElementById('overlay-desc').innerHTML = 
        `최종 점수: <b style="color:#ffd76b; font-size:16px;">${score}</b>점 · <b style="color:#4dd9ff;">${lines}</b>줄 제거<br>` + leaderboardHtml;
    document.querySelector('.big-btn').innerText = '🔄 다시 시작';
    document.getElementById('overlay').classList.add('active');
}

// ---------- 그리기 ----------
function drawCell(ctx, x, y, size, color) {
    ctx.fillStyle = color;
    ctx.fillRect(x, y, size, size);
    ctx.strokeStyle = 'rgba(255,255,255,0.25)';
    ctx.lineWidth = Math.max(1, size*0.05);
    ctx.strokeRect(x+1, y+1, size-2, size-2);
}

function draw() {
    bctx.clearRect(0,0,boardCanvas.width, boardCanvas.height);
    for (let y=0;y<ROWS;y++) {
        for (let x=0;x<COLS;x++) {
            if (grid[y][x]) drawCell(bctx, x*cellSize, y*cellSize, cellSize, grid[y][x]);
        }
    }
    let ghostY = 0;
    while (!collides(current, 0, ghostY+1)) ghostY++;
    bctx.globalAlpha = 0.25;
    current.cells.forEach(([cx,cy]) => {
        const gx = current.x+cx, gy = current.y+cy+ghostY;
        if (gy >= 0) drawCell(bctx, gx*cellSize, gy*cellSize, cellSize, current.color);
    });
    bctx.globalAlpha = 1;
    current.cells.forEach(([cx,cy]) => {
        const gx = current.x+cx, gy = current.y+cy;
        if (gy >= 0) drawCell(bctx, gx*cellSize, gy*cellSize, cellSize, current.color);
    });
    bctx.strokeStyle = 'rgba(255,255,255,0.05)';
    for (let x=0;x<=COLS;x++) { bctx.beginPath(); bctx.moveTo(x*cellSize,0); bctx.lineTo(x*cellSize,boardCanvas.height); bctx.stroke(); }
    for (let y=0;y<=ROWS;y++) { bctx.beginPath(); bctx.moveTo(0,y*cellSize); bctx.lineTo(boardCanvas.width,y*cellSize); bctx.stroke(); }
}

function drawMiniPiece(ctx, key) {
    ctx.clearRect(0,0,ctx.canvas.width, ctx.canvas.height);
    if (!key) return;
    const shape = SHAPES[key];
    const size = Math.min(ctx.canvas.width, ctx.canvas.height) / 4.4;
    const xs = shape.cells.map(c=>c[0]), ys = shape.cells.map(c=>c[1]);
    const minX = Math.min(...xs), maxX = Math.max(...xs), minY = Math.min(...ys), maxY = Math.max(...ys);
    const w = (maxX-minX+1)*size, h = (maxY-minY+1)*size;
    const offX = (ctx.canvas.width - w)/2 - minX*size;
    const offY = (ctx.canvas.height - h)/2 - minY*size;
    shape.cells.forEach(([cx,cy]) => drawCell(ctx, offX+cx*size, offY+cy*size, size, shape.color));
}

function drawHold() { drawMiniPiece(hctx, holdPiece); }
function drawNext() { drawMiniPiece(nctx, bag[0]); }

// ---------- 루프 ----------
function loop(t) {
    if (!running) return;
    const dt = t - lastTime;
    lastTime = t;
    dropTimer += dt;
    elapsedPlayMs += dt;
    
    // 퀴즈용 시간 누적 및 체크
    quizTimerMs += dt;
    if (quizTimerMs >= nextQuizTimeMs) {
        triggerQuiz();
        return; // 퀴즈 호출 시 일시 정지되므로 함수 종료
    }

    timeBoostLevel = Math.floor(elapsedPlayMs / TIME_BOOST_INTERVAL);
    let effInterval = dropInterval - timeBoostLevel * 40;
    if (elapsedPlayMs < slowEndMs) effInterval *= 1.6;
    effInterval = Math.max(90, effInterval);
    if (dropTimer > effInterval) {
        dropTimer = 0;
        if (!collides(current, 0, 1)) current.y++;
        else lockPiece();
    }
    draw();
    requestAnimationFrame(loop);
}

let touchStartX=null, touchStartY=null, touchMoved=false;
boardCanvas.addEventListener('touchstart', (e)=>{ const t=e.touches[0]; touchStartX=t.clientX; touchStartY=t.clientY; touchMoved=false; }, {passive:true});
boardCanvas.addEventListener('touchmove', (e)=>{
    if (touchStartX===null) return;
    const t = e.touches[0];
    const dx = t.clientX-touchStartX, dy = t.clientY-touchStartY;
    if (Math.abs(dx) > cellSize*0.8) {
        if (dx > 0) moveRight(); else moveLeft();
        touchStartX = t.clientX; touchMoved = true;
    } else if (dy > cellSize*0.9) {
        softDrop();
        touchStartY = t.clientY; touchMoved = true;
    }
    e.preventDefault();
}, {passive:false});
boardCanvas.addEventListener('touchend', ()=>{ if (!touchMoved) rotatePiece(); touchStartX=null; touchStartY=null; });

    else if (e.key === 'ArrowRight') moveRight();
    else if (e.key === 'ArrowDown') softDrop();
    else if (e.key === 'ArrowUp') rotatePiece();
    else if (e.code === 'Space') hardDrop();
    else if (e.key === 'c' || e.key === 'C') doHold();
});
